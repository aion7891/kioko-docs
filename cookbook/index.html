    Cookbook¶

    From the source repository

    Every recipe's full code and real output are on this page: copy one into a file and run it after pip install kioko. File names such as examples/cookbook/01_profile.py refer to the source repository, which is not public.

    Runnable recipes for every kind of memory, the intelligence, the stores and the service. Every recipe runs offline (python examples/cookbook/<file>.py from the repository root), and the output shown under each one is real: this page is generated from the files by python docs/build_cookbook.py (ids, ports and timings are replaced by placeholders). tests/test_cookbook.py runs every recipe on every test run.

    The memory kinds¶

    01 Profile memory¶

    Who the user is - one value per slot, older values kept as history.

    examples/cookbook/01_profile.py

    from kioko import MemoryLayer

memory = MemoryLayer()
user = "robin"

memory.profile.set("name", "Robin Collins", user_id=user)
memory.profile.set("role", "senior AI engineer", user_id=user)
memory.profile.set("employer", "Fabrikam Automation", user_id=user)
memory.profile.set("location", "Oslo", user_id=user)
memory.profile.set("location", "Edinburgh", user_id=user)    # moved: the slot is updated, not duplicated

print(memory.profile.get(user_id=user))
print("location history:", [h["content"] for h in memory.profile.history("location", user_id=user)])
print(memory.profile.render(user_id=user))

# Conversations fill the same slots automatically
memory.learn("By the way, I'm now based in Rotterdam.", user_id=user)
print("after learn():", memory.profile.get("location", user_id=user))

    Output:

    {'employer': 'Fabrikam Automation', 'location': 'Edinburgh', 'name': 'Robin Collins', 'role': 'senior AI engineer'}

    02 Factual memory¶

    Facts and preferences, with confidence and verification.

    examples/cookbook/02_facts.py

    from kioko import MemoryLayer

memory = MemoryLayer()
user = "robin"

memory.facts.add("The Oslo plant runs CoreLogic 5580 controllers", user_id=user, confidence=0.7,
                 topics=["plant"])
memory.facts.add("User prefers answers with code first", user_id=user, preference=True)
manager = memory.facts.add("User's manager is Natalie", user_id=user, key="work.manager")

memory.facts.verify(manager, user_id=user, confidence=0.99)           # confirmed by the user / a system of record

for record in memory.facts.list(user_id=user):
    print(f"[{record.memory_type}] {record.content}  {record.metadata}")

hits = memory.facts.search("which controllers does the plant use?", user_id=user)
print("best match:", hits[0].record.content, round(hits[0].score, 2))

# Contradictions supersede instead of piling up
memory.add("User likes long explanations", user_id=user, memory_type="preference")
memory.add("User doesn't like long explanations", user_id=user, memory_type="preference")
print([r.content for r in memory.facts.list(user_id=user) if "explanations" in r.content])

    Output:

    [fact] User's manager is Natalie  {'verified': True, 'confidence': 0.99}
[preference] User prefers answers with code first  {}
[fact] The Oslo plant runs CoreLogic 5580 controllers  {'confidence': 0.7}
best match: The Oslo plant runs CoreLogic 5580 controllers 0.52
["User doesn't like long explanations"]

    03 Semantic memory¶

    Domain knowledge - shared by every user, or personal.

    examples/cookbook/03_semantic.py

    from kioko import MemoryLayer

memory = MemoryLayer()

# A shared knowledge base (stored under the reserved user "@shared")
memory.semantic.add("OPC UA servers listen on TCP port 4840 by default", topic="opc-ua")
memory.semantic.ingest(
    """Nimbus HMI Studio is an HMI and IIoT platform. It connects to OPC UA servers.

    Nimbus projects publish tag data over MQTT. Use the cloud connector for remote dashboards.""",
    source="nimbus-handbook", topic="hmi")

# Personal knowledge for one user
memory.semantic.add("Robin's test rig uses port 4841", user_id="robin", topic="opc-ua")

for hit in memory.semantic.search("which port does OPC UA use?", user_id="robin"):
    print(f"{hit.score:.2f}  [{hit.record.user_id}] {hit.record.content}")

print("topics:", memory.semantic.topics())

# Re-ingesting a document replaces its chunks; forget_source() removes it entirely
memory.semantic.ingest("Nimbus HMI Studio 2.0 adds a Python runtime.", source="nimbus-handbook", topic="hmi")
print("handbook now:", [r.content for r in memory.semantic.list() if (r.key or "").startswith("knowledge:nimbus")])

    04 Episodic memory¶

    What happened, when, where and with whom - consolidated as it ages.

    examples/cookbook/04_episodic.py

    import time

from kioko import MemoryLayer

DAY = 86400
memory = MemoryLayer()
user = "robin"

memory.episodes.record("Presented the predictive-maintenance demo to the plant team", user_id=user,
                       where="Oslo plant", who=["Natalie", "plant team"], outcome="approved for a pilot", valence=0.8)
for line in (1, 2, 3):
    memory.episodes.record(f"Calibrated the vision sensor on packaging line {line}", user_id=user,
                           when=time.time() - (60 + line) * DAY, where="Oslo plant", outcome="within tolerance")

for episode in memory.episodes.timeline(user_id=user):
    print(f"{time.strftime('%Y-%m-%d', time.gmtime(episode.occurred_at))}  {episode.what}  -> {episode.outcome}")

print("last 7 days:", [e.what for e in memory.episodes.recent(user_id=user, days=7)])

# Old, similar episodes are folded into one summary memory (and kept, at half importance)
print(memory.episodes.consolidate(user_id=user, older_than_days=30))
print([r.content[:90] for r in memory.list(user_id=user, memory_types=["summary"])])

# Conversation turns with a time reference become episodes automatically
memory.learn("Yesterday I met Natalie to review the architecture.", user_id=user)
print([e.what for e in memory.episodes.recent(user_id=user, days=2)])

    05 Working memory¶

    The current task's scratchpad - per session, small, expiring, never embedded.

    examples/cookbook/05_working.py

    from kioko import MemoryLayer

memory = MemoryLayer()
wm = memory.working("ticket-4711", user_id="robin", capacity=4)

wm.set("task", "Find why line 3's OPC UA client disconnects every hour")
wm.set("hypothesis", "session timeout shorter than the publishing interval")
wm.note("Server log shows BadSessionIdInvalid at 14:00, 15:00, 16:00")
wm.set("next_step", "compare the client's session timeout with the server's")

print(wm.render())

# The scratchpad is part of the agent's context for this session (and only this session)
print(memory.context("what should I check next?", user_id="robin", session_id="ticket-4711").text)

# Keep a conclusion for the long term, drop the rest when the task is done
wm.set("finding", "Client session timeout was 60 minutes; the server closes idle sessions after 59")
fact = wm.promote("finding")
print("promoted:", fact.memory_type, "-", fact.content)
print("cleared:", wm.clear(), "items")

    06 Conversational memory¶

    Rolling summary + the last few turns - constant size however long the chat gets.

memory = MemoryLayer()
chat = memory.conversation("support-chat-1", user_id="robin", keep_last=2, summarize_after=8)

turns = [
    ("The packaging line stopped with fault E-217.", "E-217 is a servo overtravel. Check the limit switches."),
    ("Limit switches look fine.", "Then check the encoder cable on axis 2."),
    ("The encoder cable was loose; I reseated it.", "Good. Clear the fault and jog the axis slowly."),
    ("Axis 2 jogs fine now.", "Run one cycle at 20% speed."),
    ("One cycle completed.", "Restore production speed and watch the following error."),
    ("Following error stays under 0.1 mm.", "The line is back to normal."),
]
for user_text, reply in turns:
    chat.add_turn(user_text, reply)

print("stats:", chat.stats())
print("summary:", chat.summary())
print()
print(chat.context(max_tokens=200))
print()
print([m["role"] for m in chat.as_messages(system="You are a maintenance assistant.")])

memory = MemoryLayer()

    08 Procedural memory¶

    How-tos with versioned steps and a success record, plus standing rules.

    examples/cookbook/08_procedural.py

    from kioko import MemoryLayer

memory = MemoryLayer()
user = "robin"

memory.procedures.add("deploy the agent to the edge gateway", user_id=user,
                      steps=["build the container image", "push it to the plant registry",
                             "roll it out with the fleet manager"],
                      when_to_use="releasing a new agent version", tools=["docker", "fleet-manager"])
memory.procedures.record_outcome("deploy the agent to the edge gateway", success=True, user_id=user)

# A better way is found: same name, new steps -> version 2 (success counts carry over)
v2 = memory.procedures.add("deploy the agent to the edge gateway", user_id=user,
                           steps=["build the container image", "run the smoke tests", "push it to the plant registry",
                                  "roll out to one gateway, then the fleet"])
print(v2.render(), f"\n(version {v2.version}, success rate {v2.success_rate})")

for procedure, score in memory.procedures.find("how do I release a new version of the agent?", user_id=user):
    print(f"{score:.2f}  {procedure.name}")

# Behavioural rules are procedural memory too (they are always in the context)
memory.procedures.add_rule("Always run the smoke tests before a rollout", user_id=user)
print([r.content for r in memory.procedures.rules(user_id=user)])

# Described in conversation -> stored as a procedure
memory.learn("To reset the safety relay, switch off the main breaker, then press reset for 5 seconds, "
             "then switch the breaker back on.", user_id="robin")
print(memory.procedures.get("reset the safety relay", user_id=user).steps)

    Output:

    How to deploy the agent to the edge gateway (use when releasing a new agent version):
  1. build the container image
  2. run the smoke tests
  3. push it to the plant registry
  4. roll out to one gateway, then the fleet

    10 Strategy memory¶

memory = MemoryLayer()

    11 Decision memory¶

    ADR-style decisions with context, options, rationale, outcome and supersession.

    from kioko import MemoryLayer

memory = MemoryLayer()

    12 Graph memory¶

    A temporal knowledge graph - relations with history, multi-hop queries, entity-aware recall.

    import time

memory = MemoryLayer(auto_graph=True)       # also learn relations from conversations

graph.add("Robotics architecture", "part_of", "Fabrikam Automation", user_id=user)

print("now:", [r.object for r in graph.relations(user_id=user, subject="Robin", predicate="works_at")])

print("path:", " / ".join(r.line() for r in graph.path("Robin", "Robotics architecture", user_id=user)))

    13 KV-cache memory¶

    Keep the provider's prompt cache warm and never compute the same thing twice.

    examples/cookbook/13_kv_cache.py

    from kioko import MemoryLayer

memory = MemoryLayer()
user = "robin"
memory.add("Always answer with code first", user_id=user, memory_type="instruction")
memory.profile.set("role", "senior AI engineer", user_id=user)
memory.goals.add("Ship the Kioko memory layer v1", user_id=user, priority="high")
memory.facts.add("The Oslo plant runs CoreLogic controllers", user_id=user)

# 1) The stable prefix (instructions, profile, goals) is byte-identical across turns -> provider cache hits.
for _ in range(3):
    prefix = memory.kv.prefix(user_id=user)
    print(prefix.hash, prefix.tokens, "tokens, reused:", prefix.reused)

# 2) Request layouts with the cacheable part first.
system = memory.kv.anthropic_system("Which controllers does the plant use?", user_id=user,
                                    base_system="You are a plant support agent.", ttl="1h")
print([("cache_control" in block, block["text"][:40]) for block in system])

    lines = [line for line in system.splitlines() if line.startswith("- ")]

    21 Model-driven memory management¶

    One LLM call per turn decides every memory operation, and Kioko's algorithms skip trivial turns (no call), validate what the model proposes, and take over if the model fails.

    Any prompt -> text function works. With OpenAI / Azure OpenAI it is a few lines:

    from openai import OpenAI                       # or AzureOpenAI(azure_endpoint=..., api_version=...)
client = OpenAI()

def ask(prompt: str) -> str:
    reply = client.chat.completions.create(model=os.environ["KIOKO_MODEL"], temperature=0,
                                           response_format={"type": "json_object"},
                                           messages=[{"role": "user", "content": prompt}])
    return reply.choices[0].message.content

memory = kioko.intelligent("sqlite:///memory.db", llm=ask)

# A stand-in for a semantic embedding model. Like real ones, it gives unrelated texts a baseline similarity, so
# every memory competes for a place in the context - the situation in which the plan's caps matter. (With the
# default lexical embedder retrieval is already narrow and planning mostly changes *what* is retrieved.)
_lexical = HashingEmbedder()


def semantic_like(text):
    vector = [0.8 * x for x in _lexical.embed_one(text)]
    vector[0] += 0.6
    norm = math.sqrt(sum(x * x for x in vector))
    return [x / norm for x in vector]

    23 Memory graph¶

    Every memory is linked to the memories it relates to - a decision serves a goal, a fix resolves an incident, a lesson refines a procedure - with the evidence for each link. Recall follows the links and tells the agent how things connect. The algorithm is described in docs/graph-memory.md.

    examples/cookbook/23_memory_graph.py

    import kioko

memory = kioko.intelligent()          # linking is on with the intelligence (or MemoryLayer(auto_link=True))
user = "robin"

# 1. A goal, and one fact in the entity graph: Kioko uses pgvector.
goal = memory.goals.add("Ship the Kioko memory layer by December 15", user_id=user)
memory.graph.add("Kioko", "uses", "pgvector", user_id=user)

# 2. A decision mentions pgvector, not Kioko - the entity graph bridges them.
report = memory.manage("We decided to use pgvector because we already run Postgres.", user_id=user)
print(report.summary())

# 3. An incident: a fault, then its fix, in one session. Neither says "resolves"; the algorithm sees a problem and
#    a fix about the same asset in the same session.
memory.procedures.add("Clear a conveyor jam on line 3", user_id=user,
                      steps=["Press stop", "Open the guard", "Remove the jammed bottle", "Reset the drive"])
for text in ("The line 3 conveyor jammed twice this morning.", "Line 3 keeps faulting with error E-217.",
             "Found it: the encoder cable on line 3 was loose."):
    memory.manage(text, user_id=user, session_id="shift-1")
memory.strategies.reflect("clearing conveyor jams", "reset the drive before reopening the guard", "success",
                          lesson="always check the photo-eye alignment after a jam", user_id=user)

# 4. Every link carries its type, weight, author and evidence.
print("\nlinks of the goal:")
for link in memory.graph.links(goal.id):
    print(f"  {link.line()}  [{link.weight:.2f} {link.by}]  {link.why}")
fault = next(r for r in memory.list(user_id=user, memory_types=["episodic"]) if "E-217" in r.content)
print("\nlinks of the fault:")
for link in memory.graph.links(fault.id):
    print(f"  {link.line()}  [{link.weight:.2f} {link.by}]  {link.why}")

# 5. Why are two memories (not) linked? explain() shows the evidence the algorithm used.
tea = memory.add("User likes green tea.", user_id=user)
verdict = memory.graph.explain(tea.id, fault.id)
print("\ntea vs fault:", verdict.linked, {k: round(v, 2) for k, v in verdict.signals.items() if v})

# 6. Spreading activation: what is connected to the lesson, best first, and through which links.
lesson = memory.strategies.list(user_id=user)[0]
print("\nrelated to the lesson:")
for item in memory.graph.related(lesson.id, depth=2, limit=4):
    print(f"  {item.score:.2f}  {item.record.content[:60]}")
    for step in item.path:
        print(f"        via {step.line()}")

# 7. Recall follows the links and says how the memories connect.
ctx = memory.recall("Line 3 is faulting with E-217 again - what do we know?", user_id=user)
print()
print(ctx.text)
print("brought in by links:", [r.record.content[:40] for section in ("procedures", "strategies", "relevant")
                               for r in getattr(ctx, section) if "linked" in r.signals])

# 8. Link by hand (never fades), trace a chain, and draw the graph.
link = memory.graph.link(lesson.id, fault.id, type="learned_from", why="the E-217 night taught us this")
print("\nmanual:", link.line(), link.by)
print("trace lesson -> fault:", " | ".join(step.line() for step in memory.graph.trace(lesson.id, fault.id)))
graph = memory.graph.memories(user_id=user)
print("graph:", graph["stats"])

# 9. Memories written before linking was on are linked in one call.
legacy = kioko.MemoryLayer(auto_link=False)
legacy.add("The OPC UA server on the HMI panel rejected our client certificate.", user_id=user,
           memory_type="episodic")
legacy.strategies.reflect("debugging OPC UA", "restart the server", "failure",
                          lesson="check the OPC UA certificate trust list first", user_id=user)
print("\nconnect():", {k: v for k, v in legacy.graph.connect(user_id=user).items() if k != "duration_ms"})
